import { describe, it, expect, vi, beforeEach, afterEach } from 'vitest';
import { render, screen } from '@testing-library/react';
import { App } from '../src/App.jsx';

const productsResponse = (products) => ({
  ok: true, status: 200, json: async () => ({ products }),
});

beforeEach(() => { vi.stubGlobal('fetch', vi.fn()); });
afterEach(() => { vi.unstubAllGlobals(); vi.restoreAllMocks(); });

describe('App', () => {
  it('renders the price the service actually sent, formatted', async () => {
    fetch.mockResolvedValue(productsResponse([
      { id: '1', slug: 'lamp', name: 'Desk lamp', priceCents: 4500, stock: 3 },
    ]));
    render(<App />);
    // findBy* because the list appears after the request resolves.
    expect(await screen.findByText('Desk lamp')).toBeInTheDocument();
    expect(screen.getByText('$45.00')).toBeInTheDocument();
    expect(screen.getByText('In stock')).toBeInTheDocument();
  });

  it('shows an empty state rather than an empty page', async () => {
    fetch.mockResolvedValue(productsResponse([]));
    render(<App />);
    expect(await screen.findByText('No products yet.')).toBeInTheDocument();
  });

  it('shows the failure instead of rendering nothing', async () => {
    fetch.mockResolvedValue({ ok: false, status: 503, json: async () => ({}) });
    render(<App />);
    expect(await screen.findByRole('alert')).toHaveTextContent('503');
  });

  it('marks a sold-out product', async () => {
    fetch.mockResolvedValue(productsResponse([
      { id: '2', slug: 'chair', name: 'Chair', priceCents: 12000, stock: 0 },
    ]));
    render(<App />);
    expect(await screen.findByText('Sold out')).toBeInTheDocument();
  });
});
