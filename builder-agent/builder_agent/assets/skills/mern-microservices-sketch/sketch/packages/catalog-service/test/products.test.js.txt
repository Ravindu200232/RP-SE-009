import { describe, it, expect, beforeAll, afterAll, beforeEach } from 'vitest';
import request from 'supertest';
import { createApp } from '../src/app.js';
import { connectDb, disconnectDb } from '../src/db.js';
import { Product } from '../src/models/product.js';

const uri = process.env.MONGODB_URI ?? 'mongodb://127.0.0.1:27017/sketch_mern_test';
const app = createApp();

beforeAll(async () => { await connectDb(uri); });
afterAll(async () => { await disconnectDb(); });
// Between tests, not once per file: a document left behind changes the next
// test's result and the failure reads as a product bug.
beforeEach(async () => { await Product.deleteMany({}); });

describe('GET /products', () => {
  it('returns the public shape, sorted by name', async () => {
    await Product.create([
      { slug: 'lamp', name: 'Desk lamp', priceCents: 4500, stock: 3 },
      { slug: 'chair', name: 'Chair', priceCents: 12000, stock: 1 },
    ]);
    const response = await request(app).get('/products').expect(200);
    expect(response.body.products.map((product) => product.slug)).toEqual(['chair', 'lamp']);
    // The field the client renders has to cross the boundary, not merely exist
    // in the model.
    expect(response.body.products[0]).toEqual({
      id: expect.any(String), slug: 'chair', name: 'Chair', priceCents: 12000, stock: 1,
    });
  });

  it('filters by name, case-insensitively', async () => {
    await Product.create([
      { slug: 'lamp', name: 'Desk lamp', priceCents: 4500, stock: 3 },
      { slug: 'chair', name: 'Chair', priceCents: 12000, stock: 1 },
    ]);
    const response = await request(app).get('/products?q=desk').expect(200);
    expect(response.body.products).toHaveLength(1);
    expect(response.body.products[0].slug).toBe('lamp');
  });
});

describe('GET /products/:slug', () => {
  it('404s for an unknown slug', async () => {
    const response = await request(app).get('/products/missing').expect(404);
    expect(response.body).toEqual({ error: 'Product not found' });
  });
});

describe('POST /products/:slug/reserve', () => {
  it('decrements stock', async () => {
    await Product.create({ slug: 'lamp', name: 'Desk lamp', priceCents: 4500, stock: 3 });
    const response = await request(app).post('/products/lamp/reserve').send({ quantity: 2 }).expect(200);
    expect(response.body.product.stock).toBe(1);
  });

  it('rejects a reservation larger than stock and leaves it unchanged', async () => {
    await Product.create({ slug: 'lamp', name: 'Desk lamp', priceCents: 4500, stock: 1 });
    await request(app).post('/products/lamp/reserve').send({ quantity: 2 }).expect(409);
    expect((await Product.findOne({ slug: 'lamp' })).stock).toBe(1);
  });

  it('never lets concurrent reservations drive stock below zero', async () => {
    await Product.create({ slug: 'lamp', name: 'Desk lamp', priceCents: 4500, stock: 5 });
    const results = await Promise.all(Array.from({ length: 8 }, () =>
      request(app).post('/products/lamp/reserve').send({ quantity: 1 })));
    expect(results.filter((result) => result.status === 200)).toHaveLength(5);
    expect((await Product.findOne({ slug: 'lamp' })).stock).toBe(0);
  });

  it('rejects a non-positive quantity', async () => {
    await Product.create({ slug: 'lamp', name: 'Desk lamp', priceCents: 4500, stock: 5 });
    await request(app).post('/products/lamp/reserve').send({ quantity: 0 }).expect(400);
  });
});
