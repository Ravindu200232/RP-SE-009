// @vitest-environment node
// Data access runs in Node. Putting it in jsdom to share one config hides the
// environment the code actually runs in.
import { describe, it, expect, beforeAll, afterAll, beforeEach } from 'vitest';
import { connectDb, disconnectDb } from '@/lib/db.js';
import { Product } from '@/models/Product.js';
import { listProducts, findProduct } from '@/lib/products.js';

const uri = process.env.MONGODB_URI ?? 'mongodb://127.0.0.1:27017/sketch_next_test';

beforeAll(async () => { await connectDb(uri); });
afterAll(async () => { await disconnectDb(); });
beforeEach(async () => { await Product.deleteMany({}); });

describe('the product data layer', () => {
  it('returns the public shape, sorted by name', async () => {
    await Product.create([
      { slug: 'lamp', name: 'Desk lamp', priceCents: 4500, stock: 3 },
      { slug: 'chair', name: 'Chair', priceCents: 12000, stock: 1 },
    ]);
    const products = await listProducts();
    expect(products.map((item) => item.slug)).toEqual(['chair', 'lamp']);
    expect(products[0]).toEqual({ id: expect.any(String), slug: 'chair', name: 'Chair', priceCents: 12000, stock: 1 });
  });

  it('filters case-insensitively', async () => {
    await Product.create([
      { slug: 'lamp', name: 'Desk lamp', priceCents: 4500, stock: 3 },
      { slug: 'chair', name: 'Chair', priceCents: 12000, stock: 1 },
    ]);
    expect(await listProducts('desk')).toHaveLength(1);
  });

  it('returns null for a missing slug rather than throwing', async () => {
    expect(await findProduct('missing')).toBeNull();
  });

  it('rejects a duplicate slug with the duplicate-key error, not a validation error', async () => {
    await Product.create({ slug: 'lamp', name: 'Desk lamp', priceCents: 4500, stock: 3 });
    await Product.init(); // let the unique index finish building before asserting on it
    await expect(Product.create({ slug: 'lamp', name: 'Another lamp', priceCents: 100, stock: 1 }))
      .rejects.toMatchObject({ code: 11000 });
  });
});
