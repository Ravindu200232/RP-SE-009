import { describe, it, expect } from 'vitest';
import { render, screen } from '@testing-library/react';
import { ProductList } from '@/components/ProductList.jsx';

const product = (over = {}) => ({ id: '1', slug: 'lamp', name: 'Desk lamp', priceCents: 4500, stock: 3, ...over });

describe('ProductList', () => {
  it('renders the formatted price, not the raw field', () => {
    render(<ProductList products={[product()]} />);
    expect(screen.getByText('Desk lamp')).toBeInTheDocument();
    expect(screen.getByText('$45.00')).toBeInTheDocument();
  });

  it('links each product to its own page', () => {
    render(<ProductList products={[product()]} />);
    expect(screen.getByRole('link', { name: 'Desk lamp' })).toHaveAttribute('href', '/products/lamp');
  });

  it('says so when there is nothing, instead of rendering nothing', () => {
    render(<ProductList products={[]} />);
    expect(screen.getByText('No products yet.')).toBeInTheDocument();
  });

  it('marks a sold-out product', () => {
    render(<ProductList products={[product({ stock: 0 })]} />);
    expect(screen.getByText('Sold out')).toBeInTheDocument();
  });
});
